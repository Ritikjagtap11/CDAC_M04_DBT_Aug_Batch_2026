# function using calculate_grade

DELIMITER //
CREATE FUNCTION calculate_grade(s_marks INT)
RETURNS VARCHAR(10)
DETERMINISTIC
NO SQL
BEGIN
	IF s_marks >= 90 THEN
	RETURN 'A+';
	
	ELSEIF s_marks >= 80 THEN
	RETURN 'A';

	ELSEIF s_marks >= 70 THEN
	RETURN 'B';
	
	ELSEIF s_marks >= 60 THEN
	RETURN 'C';
	
	ELSEIF s_marks >= 50 THEN
	RETURN 'D';
	
	ELSE
	RETURN 'F';
	
	END IF;
END //
DELIMITER ;

SELECT calculate_grade(80);

#using table 
SELECT 
	student_id,
	name,
	marks,
	calculate_grade(marks) As grade
FROM student;

#using where
SELECT 
	student_id,
	name,
	marks,
	calculate_grade(marks)
FROM student
WHERE calculate_grade(marks) = 'A';

#Trigger


CREATE TABLE audit_table(
	audit_id INT PRIMARY KEY AUTO_INCREMENT,
	emp_id INT,
	emp_name VARCHAR(30),
	old_salary DECIMAL(10,2),
	new_salary DECIMAL(10,2),
	change_on datetime
);


->after update

DELIMITER //
CREATE TRIGGER trgt_salary_audit
AFTER UPDATE ON employee
FOR EACH ROW
BEGIN
	INSERT INTO audit_table
	(emp_id,emp_name,old_salary,new_salary,change_on)
	VALUES
	(NEW.emp_id,NEW.emp_name,OLD.salary,NEW.salary,now());
END //
DELIMITER ;
