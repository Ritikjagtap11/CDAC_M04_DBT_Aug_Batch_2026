CREATE PROCEDURE apply_salary_increment()
BEGIN 
	DECLARE done BOOLEAN DEFAULT false;
	DECLARE v_emp_id INT;
	DECLARE v_emp_name VARCHAR(30);
	DECLARE v_salary DECIMAL(10,2);
	DECLARE v_grade CHAR(1);

	DECLARE cur_employee CURSOR FOR
	SELECT 	emp_id,emp_name,salary,grade
	FROM employee;
	
	DECLARE CONTINUE HANDLER FOR NOT FOUND
	SET DONE = true;

	OPEN cur_employee;

	salary_loop : loop
	
	FETCH cur_employee
	into v_emp_id,v_emp_name,v_salary,v_grade;
	
	
	IF done THEN
	 LEAVE salary_loop;
	END IF;	
	
	IF v_grade = 'A' THEN
	UPDATE employee
	SET salary = salary * 1.05
	WHERE emp_id = v_emp_id;
	
	ELSEIF v_grade = 'B' THEN
	UPDATE employee
	SET salary = salary * 1.10
	WHERE emp_id = v_emp_id;

	ELSEIF v_grade = 'C' THEN
	UPDATE employee
	SET salary = salary * 1.15
	WHERE emp_id = v_emp_id;

	ELSE 
	UPDATE employee
	SET salary = salary * 1.13
	WHERE emp_id = v_emp_id;
	END IF;
	
	END loop salary_loop;
	CLOSE cur_employee;

END //
DELIMITER ;

DELIMITER //
CREATE PROCEDURE archieve_low_salary()
BEGIN
	DECLARE done BOOLEAN DEFAULT false;

	DECLARE v_emp_id INT;
	DECLARE v_emp_name VARCHAR(30);
	DECLARE v_salary INT;
	DECLARE v_grade CHAR(1);

	DECLARE cur_employee CURSOR FOR
	SELECT emp_id, emp_name, salary, grade
	FROM employee;
	
	DECLARE CONTINUE HANDLER FOR NOT FOUND
	SET done = true;

	OPEN cur_employee;
	
	archieve_loop: LOOP
	
	FETCH cur_employee
	INTO v_emp_id, v_emp_name, v_salary, v_grade;
		
	IF done THEN
	  LEAVE archieve_loop;
	END IF;

	IF v_salary <= 50000 THEN
	INSERT INTO employee_archieve
	(emp_id,emp_name,salary,grade,acrchieve_on)
	VALUES
	(v_emp_id,v_emp_name,v_salary,v_grade,now());
	END IF;
	
	END LOOP;
	
	CLOSE cur_employee;
END //
DELIMITER ;
