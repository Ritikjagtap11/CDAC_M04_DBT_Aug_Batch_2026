Q2->
SELECT d.dep_name, AVG(e.emp_salary) as avg_salary
FROM employee e,department d
WHERE e.dep_id = d.dep_id
GROUP BY d.dep_name;

Q3->
SELECT d.dep_name, COUNT(e.emp_name) emp_count
FROM employee e, department d
WHERE e.dep_id = d.dep_id
GROUP BY d.dep_name;

Q4->
SELECT d.dep_name, COUNT(e.emp_name) emp_count
FROM employee e, department d
WHERE e.dep_id = d.dep_id
GROUP BY d.dep_name
HAVING emp_count > 5;
